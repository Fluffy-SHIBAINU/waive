from dbload import D, ref, sources
from candidate_final import names_hospital_or_town
from waive.atlas.structure import document_states
from sheet_fields import walk
for scope in ("published", "draft", "held"):
    n = 0
    for ccn, (v, st, body) in sorted(D["latest"].items()):
        if st != scope: continue
        h = ref(ccn)
        cited = collections.defaultdict(list)
        for p, val, sid in walk(json.loads(body)): cited[sid].append(p.split(".")[-1])
        for sid, fields in cited.items():
            d = D["docs"].get(sid)
            if not d or d["kind"] != "hospital_web": continue
            named = document_states(d["title"], d["url"] or "", h)
            if named and h.state not in named:
                n += 1
                print(scope, ccn, h.state, h.name[:30], sorted(named), "namesH" if names_hospital_or_town(h, d["text"]) else "-", "|", d["title"][:60], "|", (d["url"] or "")[:70], "|", ",".join(fields)[:70])
    print(scope, n)
